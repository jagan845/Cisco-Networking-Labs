Router R3 Configuration
Router> enable
Router# configure terminal
# Set hostname
Router(config)# hostname R3
# Configure FastEthernet interface
R3(config)# interface fastEthernet0/0
R3(config-if)# ip address 192.168.30.1 255.255.255.0
R3(config-if)# no shutdown
R3(config-if)# exit
# Configure Serial 0/0/0
R3(config)# interface serial0/0/0
R3(config-if)# ip address 172.18.0.1 255.255.0.0
R3(config-if)# encapsulation ppp
R3(config-if)# clock rate 64000
R3(config-if)# no shutdown
R3(config-if)# exit
# Configure Serial 0/0/1
R3(config)# interface serial0/0/1
R3(config-if)# ip address 172.17.0.2 255.255.0.0
R3(config-if)# encapsulation ppp
R3(config-if)# no shutdown
R3(config-if)# exit
# Enable IP routing and add static routes
R3(config)# ip routing
R3(config)# ip route 192.168.10.0 255.255.255.0 172.18.0.2
R3(config)# ip route 192.168.20.0 255.255.255.0 172.17.0.1
R3(config)# ip route 172.16.0.0 255.255.0.0 172.17.0.1
R3(config)# end
R3# write
